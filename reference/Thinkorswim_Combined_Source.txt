# ============================================================
# DUAL MARKET DECISION DASHBOARD - S&P 500 + NASDAQ-100
# Based on user's last-working 16-point SPY study (21 Nov 2026)
#
# FIVE SECTIONS (individual colored labels):
#   1 SP500 SCORE / 5D TREND / CONFIRMATION / SWING ENV
#   2 SPY-specific internals (VIX, SPX P/C, SPY leadership/breadth)
#   3 Shared macro (oil, USD, TNX, credit, copper, VIX curve)
#   4 NASDAQ SCORE / 5D TREND / CONFIRMATION / SWING ENV
#   5 Nasdaq internals (VXN, PCND, QQQ leadership, NDX A/D breadth)
#
# EACH SCORE HAS 15 POINTS FOR DIRECT COMPARISON:
# SP500: original 16 scored signals minus VXN (moved to Nasdaq).
# Nasdaq: same six shared macro signals + nine Nasdaq-specific signals.
# Shared macro is displayed ONLY ONCE, but used by BOTH scores.
# Red +1; yellow/green 0. Missing data never counts as bullish.
# Both 0-4 RISK-ON, 5-7 CAUTION, 8-15 RISK-OFF.
#
# IMPORTANT DIFFERENCE: SP500 breadth uses changes in $SPXA200R
# (percentage of SPX above 200D SMA). Nasdaq breadth uses Nasdaq-100
# ADVANCERS/DECLINERS ($ADVND/$DECLND) and a 5D smoothed trend.
# A Nasdaq-100 $NDXA200R Thinkorswim quote was NOT verified.
#
# For five SECTION GROUPS with colored individual labels, Thinkorswim
# needs row-ownership headers between groups; this creates extra
# physical rows and labels may still wrap depending on chart width.
# It cannot guarantee exactly five screen rows with individually
# colored labels. Use SPY 1Y 1D chart for score momentum.
# ============================================================

declare upper;

input vixBullMax = 16.0;
input vixBearMin = 20.0;
input vxnBullMax = 20.0;
input vxnBearMin = 25.0;
input pcrThreshold = 1.00;
input pcrExtreme = 1.35;
input defLookbackDays = 10;
input xleLookbackDays = 10;
input smhEmaLength = 20;
input smhRatioMaLen = 15;

# Relative/price-health inputs (each red condition = 1 bearish point per applicable index)
input spyTrendFastLen = 50;
input spyTrendSlowLen = 200;
input breadth200BullMin = 60.0;
input breadth200BearMax = 40.0;
input relativeStrengthDays = 10;  # RSP/SPY and IWM/SPY ratio moving average
input relativeNeutralBandPct = 0.25;  # CAUTION if ratio is within +/-0.25% of 10D SMA
input vixCurveBullMax = 0.90;
input vixCurveBearMin = 1.00;
input vix3mSymbol = "VIX3M";

# Daily regime MOMENTUM / confirmation settings.
input scoreTrendDays = 5;          # 5 daily candles = ~one trading week
input scoreTrendMinChange = 2;     # 2+ point move = meaningful shift
input breadth1DMinPP = 1.0;       # 1D move of >=1 percentage point = directional
input breadthChangeMinPP = 2.0;    # 5D move of >=2 percentage points = directional
input breadthFeedFraction = yes;   # 0.50 feed quote means 50% of SPX above 200D MA
input showSwingEnvironment = yes;  # descriptive, not an automatic trade signal
input showSignalSectionHeader = yes; # dedicated row separators between 5 sections
input ndBreadthNeutralBand = 5.0; # NDX advancer share <45% bearish, >55% bullish
input ndBreadth5DMinPP = 5.0;     # 5-session avg advancer share shift vs prior 5 sessions
input ndPCRThreshold = 1.00;      # initial NASDAQ-100 P/C threshold; calibrate separately
input ndPCRExtreme = 1.35;        # contrarian yellow washout alert


DefineGlobalColor("BullBright", CreateColor(0, 230, 118));
DefineGlobalColor("BearBright", CreateColor(255, 60, 60));
DefineGlobalColor("WarnBright", CreateColor(255, 215, 0));
DefineGlobalColor("InfoBlue", CreateColor(0, 190, 240));
DefineGlobalColor("NeutralGray", CreateColor(130, 130, 130));

# Fixed SPY reference for macro divergences; works on either a SPY or QQQ chart.
def mktC = close("SPY", period = AggregationPeriod.DAY);
def mktP = mktC[1];
def mktPct = if !IsNaN(mktC) and !IsNaN(mktP) and mktP != 0
             then (mktC - mktP) / mktP else Double.NaN;

# S&P 500 volatility: VIX
def vixVal = close("VIX", period = AggregationPeriod.DAY);
def vixValid = !IsNaN(vixVal);
def vixBear = if !vixValid then 0 else if vixVal > vixBearMin then 1 else 0;
def vixNeut = if !vixValid then 0 else if vixVal >= vixBullMax and vixVal <= vixBearMin then 1 else 0;

# Nasdaq-100 volatility: VXN
def vxnVal = close("VXN:CGI", period = AggregationPeriod.DAY);
def vxnValid = !IsNaN(vxnVal);
def vxnBear = if !vxnValid then 0 else if vxnVal > vxnBearMin then 1 else 0;
def vxnNeut = if !vxnValid then 0 else if vxnVal >= vxnBullMax and vxnVal <= vxnBearMin then 1 else 0;

# 3. Oil vs chart: directional divergences
# Oil up = bearish; Oil down = bullish, including same-direction divergences.
def clC = close("/CL", period = AggregationPeriod.DAY);
def clChg = clC - clC[1];
def clValid = !IsNaN(clChg) and !IsNaN(mktPct);
def clBear = if !clValid then 0 else if clChg > 0 then 1 else 0;
def clBull = if !clValid then 0 else if clChg < 0 then 1 else 0;
def clBearDiv = if !clValid then 0 else if clChg > 0 and mktPct > 0 then 1 else 0;
def clBullDiv = if !clValid then 0 else if clChg < 0 and mktPct < 0 then 1 else 0;

# 4. Dollar: /DX, fallback UUP
def dxC = close("/DX", period = AggregationPeriod.DAY);
def uupC = close("UUP", period = AggregationPeriod.DAY);
def dxRaw = dxC - dxC[1];
def uupRaw = uupC - uupC[1];
def dxChg = if !IsNaN(dxRaw) then dxRaw else uupRaw;
def dxValid = !IsNaN(dxChg) and !IsNaN(mktPct);
# USD direction sets the bearish point, including same-direction divergences.
def dxBear = if !dxValid then 0 else if dxChg > 0 then 1 else 0;
def dxBull = if !dxValid then 0 else if dxChg < 0 then 1 else 0;
def dxBearDiv = if !dxValid then 0 else if dxChg > 0 and mktPct > 0 then 1 else 0;
def dxBullDiv = if !dxValid then 0 else if dxChg < 0 and mktPct < 0 then 1 else 0;

# 5. Ten-year Treasury yield: corrected thinkorswim symbol
def tnxC = close("TNX:CGI", period = AggregationPeriod.DAY);
def tnxChg = tnxC - tnxC[1];
def tnxValid = !IsNaN(tnxChg) and !IsNaN(mktPct);
# Yield direction sets the bearish point, including same-direction divergences.
def tnxBear = if !tnxValid then 0 else if tnxChg > 0 then 1 else 0;
def tnxBull = if !tnxValid then 0 else if tnxChg < 0 then 1 else 0;
def tnxBearDiv = if !tnxValid then 0 else if tnxChg > 0 and mktPct > 0 then 1 else 0;
def tnxBullDiv = if !tnxValid then 0 else if tnxChg < 0 and mktPct < 0 then 1 else 0;

# 6. Credit: HYG vs IEF
def hygC = close("HYG", period = AggregationPeriod.DAY);
def iefC = close("IEF", period = AggregationPeriod.DAY);
def hygPct = if !IsNaN(hygC) and !IsNaN(hygC[1]) and hygC[1] != 0
             then (hygC - hygC[1]) / hygC[1] else Double.NaN;
def iefPct = if !IsNaN(iefC) and !IsNaN(iefC[1]) and iefC[1] != 0
             then (iefC - iefC[1]) / iefC[1] else Double.NaN;
def creditValid = !IsNaN(hygPct) and !IsNaN(iefPct);
def creditBear = if !creditValid then 0 else if hygPct < iefPct then 1 else 0;

# 7. Copper
def hgC = close("/HG", period = AggregationPeriod.DAY);
def hgChg = hgC - hgC[1];
def hgValid = !IsNaN(hgChg);
def hgBear = if !hgValid then 0 else if hgChg < 0 then 1 else 0;

# ONE S&P 500 BREADTH SERIES FOR ALL THREE S&P LABELS
# $SPXA200R = fraction of S&P 500 stocks above their OWN 200-day SMA.
# The feed in this installation quotes 0.462 for 46.2%; convert to 0-100.
# All three breadth labels now use exactly this normalized series.
# 1D and 5D: changes in PERCENTAGE POINTS (pp); 200D: current LEVEL.
def breadth200Raw = close("$SPXA200R", period = AggregationPeriod.DAY);
def breadth200Val = if breadthFeedFraction then breadth200Raw * 100 else breadth200Raw;
def breadth200Valid = !IsNaN(breadth200Val) and breadth200Val >= 0 and breadth200Val <= 100;
def breadth200Bull = if !breadth200Valid then 0 else if breadth200Val >= breadth200BullMin then 1 else 0;
def breadth200Bear = if !breadth200Valid then 0 else if breadth200Val < breadth200BearMax then 1 else 0;

# 1D breadth direction = daily CHANGE in percentage above the 200D SMA.
# Separate neutral band filters small daily shifts near zero.
def breadth200Prev1 = breadth200Val[1];
def breadthValid = GetAggregationPeriod() == AggregationPeriod.DAY and
                   breadth200Valid and !IsNaN(breadth200Prev1) and
                   breadth200Prev1 >= 0 and breadth200Prev1 <= 100;
def breadth1DDeltaPP = if breadthValid then breadth200Val - breadth200Prev1 else Double.NaN;
def breadthBear = if !breadthValid then 0 else if breadth1DDeltaPP <= -breadth1DMinPP then 1 else 0;
def breadthBull = if !breadthValid then 0 else if breadth1DDeltaPP >= breadth1DMinPP then 1 else 0;

# SPX put/call (S&P specific)
def pcrVal = close("$PCSP", period = AggregationPeriod.DAY);
def pcrValid = !IsNaN(pcrVal);
def pcrBear = if !pcrValid then 0 else if pcrVal > pcrThreshold then 1 else 0;
def pcrStretch = if !pcrValid then 0 else if pcrVal >= pcrExtreme then 1 else 0;

# 10. Defensive rotation: 1D caution and 10D relative trend
def spyC = close("SPY", period = AggregationPeriod.DAY);
def xlpC = close("XLP", period = AggregationPeriod.DAY);
def xlvC = close("XLV", period = AggregationPeriod.DAY);
def xluC = close("XLU", period = AggregationPeriod.DAY);
def spy1D = if !IsNaN(spyC) and !IsNaN(spyC[1]) and spyC[1] != 0 then (spyC - spyC[1]) / spyC[1] else Double.NaN;
def xlp1D = if !IsNaN(xlpC) and !IsNaN(xlpC[1]) and xlpC[1] != 0 then (xlpC - xlpC[1]) / xlpC[1] else Double.NaN;
def xlv1D = if !IsNaN(xlvC) and !IsNaN(xlvC[1]) and xlvC[1] != 0 then (xlvC - xlvC[1]) / xlvC[1] else Double.NaN;
def xlu1D = if !IsNaN(xluC) and !IsNaN(xluC[1]) and xluC[1] != 0 then (xluC - xluC[1]) / xluC[1] else Double.NaN;
def defAvg1D = (xlp1D + xlv1D + xlu1D) / 3;

def xlpOld = GetValue(xlpC, defLookbackDays);
def xlvOld = GetValue(xlvC, defLookbackDays);
def xluOld = GetValue(xluC, defLookbackDays);
def spyOld = GetValue(spyC, defLookbackDays);
def xlpTrend = if !IsNaN(xlpC) and !IsNaN(xlpOld) and xlpOld != 0 then (xlpC - xlpOld) / xlpOld else Double.NaN;
def xlvTrend = if !IsNaN(xlvC) and !IsNaN(xlvOld) and xlvOld != 0 then (xlvC - xlvOld) / xlvOld else Double.NaN;
def xluTrend = if !IsNaN(xluC) and !IsNaN(xluOld) and xluOld != 0 then (xluC - xluOld) / xluOld else Double.NaN;
def spyTrend = if !IsNaN(spyC) and !IsNaN(spyOld) and spyOld != 0 then (spyC - spyOld) / spyOld else Double.NaN;
def defAvgTrend = (xlpTrend + xlvTrend + xluTrend) / 3;
def defValid = !IsNaN(defAvg1D) and !IsNaN(spy1D) and !IsNaN(defAvgTrend) and !IsNaN(spyTrend);
def defDayCaution = if !defValid then 0 else if defAvg1D > spy1D then 1 else 0;
def defTrendBear = if !defValid then 0 else if defAvgTrend > spyTrend then 1 else 0;

# 11. Energy relative leadership: XLE vs SPY
def xleC = close("XLE", period = AggregationPeriod.DAY);
def xleOld = GetValue(xleC, xleLookbackDays);
def spyOldXle = GetValue(spyC, xleLookbackDays);
def xleTrendPct = if !IsNaN(xleC) and !IsNaN(xleOld) and xleOld != 0 then (xleC - xleOld) / xleOld else Double.NaN;
def spyTrendPct = if !IsNaN(spyC) and !IsNaN(spyOldXle) and spyOldXle != 0 then (spyC - spyOldXle) / spyOldXle else Double.NaN;
def xleValid = !IsNaN(xleTrendPct) and !IsNaN(spyTrendPct);
def xleBear = if !xleValid then 0 else if xleTrendPct > spyTrendPct then 1 else 0;

# 12. Semiconductors: SMH vs SPY
def smhD = close("SMH", period = AggregationPeriod.DAY);
def smhEma20 = ExpAverage(smhD, smhEmaLength);
def spyEma20 = ExpAverage(spyC, smhEmaLength);
def smhRatio = if !IsNaN(smhD) and !IsNaN(spyC) and spyC != 0 then smhD / spyC else Double.NaN;
def smhRatioMa = Average(smhRatio, smhRatioMaLen);
def smhValid = !IsNaN(smhD) and !IsNaN(spyC) and !IsNaN(smhEma20) and !IsNaN(spyEma20) and !IsNaN(smhRatioMa);
def smhBearDiv = if !smhValid then 0 else if smhD < smhEma20 and spyC >= spyEma20 then 1 else 0;
def smhBearTrend = if !smhValid then 0 else if smhD < smhEma20 and spyC < spyEma20 then 1 else 0;
def smhLagging = if !smhValid then 0 else if smhD >= smhEma20 and smhRatio < smhRatioMa then 1 else 0;
def smhBull = if !smhValid then 0 else if smhD >= smhEma20 and smhRatio >= smhRatioMa then 1 else 0;
def smhAnyBear = if smhBearDiv == 1 or smhBearTrend == 1 then 1 else 0;

# ============================================================
# S&P 500 ADDITIONAL METRICS: price trend, leadership, VIX curve
# Uses SPY as the fixed market benchmark on any chart.
# ============================================================

# 13. SPY trend health: green if SPY > SMA50 > SMA200;
# red if SPY < SMA200; otherwise yellow.
def spySma50 = Average(spyC, spyTrendFastLen);
def spySma200 = Average(spyC, spyTrendSlowLen);
def spyHealthValid = !IsNaN(spyC) and !IsNaN(spySma50) and !IsNaN(spySma200);
def spyHealthBull = if !spyHealthValid then 0 else if spyC > spySma50 and spySma50 > spySma200 then 1 else 0;
def spyHealthBear = if !spyHealthValid then 0 else if spyC < spySma200 then 1 else 0;

# 14. Long-term breadth was defined above in section 8.
# It is DISPLAY-ONLY; unlike BREADTH 1D, it does not change the score.

# 15. Equal-weight leadership: RSP/SPY relative to its 10-day SMA.
# Compare today's RSP/SPY ratio against the average of the last 10 daily ratios.
# Neutral band avoids flipping on tiny differences around the average.
def rspC = close("RSP", period = AggregationPeriod.DAY);
def rspRel = if !IsNaN(rspC) and !IsNaN(spyC) and spyC > 0
             then rspC / spyC else Double.NaN;
def rspRelMa10 = Average(rspRel, relativeStrengthDays);
def rspRelDiffPct = if !IsNaN(rspRel) and !IsNaN(rspRelMa10) and rspRelMa10 > 0
                    then 100 * (rspRel / rspRelMa10 - 1)
                    else Double.NaN;
def rspValid = !IsNaN(rspRelDiffPct);
def rspBull = if !rspValid then 0 else if rspRelDiffPct > relativeNeutralBandPct then 1 else 0;
def rspBear = if !rspValid then 0 else if rspRelDiffPct < -relativeNeutralBandPct then 1 else 0;

# 16. VIX term structure: VIX / VIX3M (30-day / 3-month).
# VIX3M fallback uses CGI suffix where available.
def vix3mMain = close(symbol = vix3mSymbol, period = AggregationPeriod.DAY);
def vix3mAlt = close("VIX3M:CGI", period = AggregationPeriod.DAY);
def vix3mVal = if !IsNaN(vix3mMain) then vix3mMain else vix3mAlt;
def vixCurve = if vixValid and !IsNaN(vix3mVal) and vix3mVal > 0
               then vixVal / vix3mVal else Double.NaN;
def vixCurveValid = !IsNaN(vixCurve);
def vixCurveBull = if !vixCurveValid then 0 else if vixCurve < vixCurveBullMax then 1 else 0;
def vixCurveBear = if !vixCurveValid then 0 else if vixCurve > vixCurveBearMin then 1 else 0;

# 17. Small-cap leadership: IWM/SPY relative to its 10-day SMA.
def iwmC = close("IWM", period = AggregationPeriod.DAY);
def iwmRel = if !IsNaN(iwmC) and !IsNaN(spyC) and spyC > 0
             then iwmC / spyC else Double.NaN;
def iwmRelMa10 = Average(iwmRel, relativeStrengthDays);
def iwmRelDiffPct = if !IsNaN(iwmRel) and !IsNaN(iwmRelMa10) and iwmRelMa10 > 0
                    then 100 * (iwmRel / iwmRelMa10 - 1)
                    else Double.NaN;
def iwmValid = !IsNaN(iwmRelDiffPct);
def iwmBull = if !iwmValid then 0 else if iwmRelDiffPct > relativeNeutralBandPct then 1 else 0;
def iwmBear = if !iwmValid then 0 else if iwmRelDiffPct < -relativeNeutralBandPct then 1 else 0;


# ============================================================
# NASDAQ-100-SPECIFIC INDICATORS
# ============================================================

def qqqC = close("QQQ", period = AggregationPeriod.DAY);
def qqqP = qqqC[1];
def qqqPct = if !IsNaN(qqqC) and !IsNaN(qqqP) and qqqP > 0
             then (qqqC - qqqP) / qqqP else Double.NaN;

# NDX-specific put/call ratio (NASDAQ-100 component options, not all NASDAQ stocks)
def ndPcrVal = close("$PCND", period = AggregationPeriod.DAY);
def ndPcrValid = !IsNaN(ndPcrVal);
def ndPcrBear = if !ndPcrValid then 0 else if ndPcrVal > ndPCRThreshold then 1 else 0;
def ndPcrStretch = if !ndPcrValid then 0 else if ndPcrVal >= ndPCRExtreme then 1 else 0;

# Nasdaq-100 *daily advancing issues* / *declining issues*, not % > 200D SMA.
# Share of advancers = ADV/(ADV+DECL) x 100.
def ndAdv = close("$ADVND", period = AggregationPeriod.DAY);
def ndDec = close("$DECLND", period = AggregationPeriod.DAY);
def ndBreadthShare = if !IsNaN(ndAdv) and !IsNaN(ndDec) and ndAdv >= 0 and ndDec >= 0 and ndAdv + ndDec > 0
                     then 100 * ndAdv / (ndAdv + ndDec) else Double.NaN;
def ndBreadthValid = GetAggregationPeriod() == AggregationPeriod.DAY and !IsNaN(ndBreadthShare);
def ndBreadthBear = if !ndBreadthValid then 0 else if ndBreadthShare < 50 - ndBreadthNeutralBand then 1 else 0;
def ndBreadthBull = if !ndBreadthValid then 0 else if ndBreadthShare > 50 + ndBreadthNeutralBand then 1 else 0;

# Difference between most recent 5-session average of advancing share and
# previous five-session average. This is *not* Nasdaq stocks above 200D SMA.
def ndBreadth5Avg = Average(ndBreadthShare, scoreTrendDays);
def ndBreadth5Prev = GetValue(ndBreadth5Avg, scoreTrendDays);
def ndBreadth5Valid = GetAggregationPeriod() == AggregationPeriod.DAY and
                      !IsNaN(ndBreadth5Avg) and !IsNaN(ndBreadth5Prev);
def ndBreadth5Delta = if ndBreadth5Valid then ndBreadth5Avg - ndBreadth5Prev else Double.NaN;
def ndBreadth5Up = ndBreadth5Valid and ndBreadth5Delta >= ndBreadth5DMinPP;
def ndBreadth5Down = ndBreadth5Valid and ndBreadth5Delta <= -ndBreadth5DMinPP;

# Nasdaq-100 defensive rotation: XLP/XLV/XLU average vs QQQ, 10 trading days.
def ndQQQOldDef = GetValue(qqqC, defLookbackDays);
def ndQQQDefTrend = if !IsNaN(qqqC) and !IsNaN(ndQQQOldDef) and ndQQQOldDef > 0
                    then (qqqC - ndQQQOldDef) / ndQQQOldDef else Double.NaN;
def ndDefValid = !IsNaN(defAvg1D) and !IsNaN(defAvgTrend) and !IsNaN(qqqPct) and !IsNaN(ndQQQDefTrend);
def ndDefDayCaution = if !ndDefValid then 0 else if defAvg1D > qqqPct then 1 else 0;
def ndDefBear = if !ndDefValid then 0 else if defAvgTrend > ndQQQDefTrend then 1 else 0;

# Nasdaq-100 energy leadership: XLE vs QQQ, 10 trading days.
def ndQQQOldXLE = GetValue(qqqC, xleLookbackDays);
def ndQQQXLETrend = if !IsNaN(qqqC) and !IsNaN(ndQQQOldXLE) and ndQQQOldXLE > 0
                    then (qqqC - ndQQQOldXLE) / ndQQQOldXLE else Double.NaN;
def ndXLEValid = !IsNaN(xleTrendPct) and !IsNaN(ndQQQXLETrend);
def ndXLEBear = if !ndXLEValid then 0 else if xleTrendPct > ndQQQXLETrend then 1 else 0;

# Semiconductor price health / relative breakdown: SMH vs QQQ.
def ndQQQEma20 = ExpAverage(qqqC, smhEmaLength);
def ndSMHRatio = if !IsNaN(smhD) and !IsNaN(qqqC) and qqqC > 0
                 then smhD / qqqC else Double.NaN;
def ndSMHRatioMA = Average(ndSMHRatio, smhRatioMaLen);
def ndSMHValid = !IsNaN(smhD) and !IsNaN(qqqC) and !IsNaN(smhEma20) and
                 !IsNaN(ndQQQEma20) and !IsNaN(ndSMHRatioMA);
def ndSMHBearDiv = if !ndSMHValid then 0 else if smhD < smhEma20 and qqqC >= ndQQQEma20 then 1 else 0;
def ndSMHBearTrend = if !ndSMHValid then 0 else if smhD < smhEma20 and qqqC < ndQQQEma20 then 1 else 0;
def ndSMHLagging = if !ndSMHValid then 0 else if smhD >= smhEma20 and ndSMHRatio < ndSMHRatioMA then 1 else 0;
def ndSMHBull = if !ndSMHValid then 0 else if smhD >= smhEma20 and ndSMHRatio >= ndSMHRatioMA then 1 else 0;
def ndSMHBear = if ndSMHBearDiv == 1 or ndSMHBearTrend == 1 then 1 else 0;

# Nasdaq-100 absolute trend: QQQ vs 50D / 200D SMA.
def ndQQQSma50 = Average(qqqC, spyTrendFastLen);
def ndQQQSma200 = Average(qqqC, spyTrendSlowLen);
def ndQQQHealthValid = !IsNaN(qqqC) and !IsNaN(ndQQQSma50) and !IsNaN(ndQQQSma200);
def ndQQQHealthBull = if !ndQQQHealthValid then 0 else if qqqC > ndQQQSma50 and ndQQQSma50 > ndQQQSma200 then 1 else 0;
def ndQQQHealthBear = if !ndQQQHealthValid then 0 else if qqqC < ndQQQSma200 then 1 else 0;

# Equal-weight Nasdaq-100 ETF / cap-weighted QQQ, 10-day SMA + neutral band.
def qqqeC = close("QQQE", period = AggregationPeriod.DAY);
def ndEqualRatio = if !IsNaN(qqqeC) and !IsNaN(qqqC) and qqqC > 0
                   then qqqeC / qqqC else Double.NaN;
def ndEqualMA = Average(ndEqualRatio, relativeStrengthDays);
def ndEqualDiffPct = if !IsNaN(ndEqualRatio) and !IsNaN(ndEqualMA) and ndEqualMA > 0
                     then 100 * (ndEqualRatio / ndEqualMA - 1) else Double.NaN;
def ndEqualValid = !IsNaN(ndEqualDiffPct);
def ndEqualBull = if !ndEqualValid then 0 else if ndEqualDiffPct > relativeNeutralBandPct then 1 else 0;
def ndEqualBear = if !ndEqualValid then 0 else if ndEqualDiffPct < -relativeNeutralBandPct then 1 else 0;

# IWM vs QQQ relative to its 10-day moving-average ratio.
def ndIWMRatio = if !IsNaN(iwmC) and !IsNaN(qqqC) and qqqC > 0
                then iwmC / qqqC else Double.NaN;
def ndIWMMA = Average(ndIWMRatio, relativeStrengthDays);
def ndIWMDiffPct = if !IsNaN(ndIWMRatio) and !IsNaN(ndIWMMA) and ndIWMMA > 0
                   then 100 * (ndIWMRatio / ndIWMMA - 1) else Double.NaN;
def ndIWMValid = !IsNaN(ndIWMDiffPct);
def ndIWMBull = if !ndIWMValid then 0 else if ndIWMDiffPct > relativeNeutralBandPct then 1 else 0;
def ndIWMBear = if !ndIWMValid then 0 else if ndIWMDiffPct < -relativeNeutralBandPct then 1 else 0;

# ============================================================
# TWO 15-POINT RISK SCORES
# SHARED: OIL, USD, TNX, CREDIT, COPPER, VIX CURVE (6)
# S&P ONLY: VIX, SPX P/C, DEF, XLE, SMH, SPY TREND,
#           RSP/SPY, IWM/SPY, SPX BREADTH 1D (9)
# NASDAQ ONLY: VXN, NDX P/C, DEF/QQQ, XLE/QQQ, SMH/QQQ,
#              QQQ TREND, QQQE/QQQ, IWM/QQQ, NDX A/D 1D (9)
# ============================================================

def sharedBear = clBear + dxBear + tnxBear + creditBear + hgBear + vixCurveBear;
def bearCount = sharedBear + vixBear + pcrBear + defTrendBear + xleBear +
